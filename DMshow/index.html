目录约定

  profile.json     # 名字/标签/链接
  about.md         # 长文简介
  avatar.jpg       # 头像（可选）
  works/项目名/    # 作品集
    cover.jpg
    README.md      # 摘要
    *.jpg *.mp4
  albums/相册名/   # 图集
    desc.md
    *.jpg
  videos/          # 视频
    *.mp4 或 系列文件夹/
  gallery/         # 散图画廊
    *.jpg
